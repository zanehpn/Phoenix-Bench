module clk_int_div #(
  parameter int unsigned DIV_VALUE_WIDTH = 4,
  parameter int unsigned DEFAULT_DIV_VALUE = 0,
  parameter bit          ENABLE_CLOCK_IN_RESET = 1'b0
) (
  input  logic                       clk_i,
  input  logic                       rst_ni,
  input  logic                       en_i,
  input  logic                       test_mode_en_i,
  input  logic [DIV_VALUE_WIDTH-1:0]  div_i,
  input  logic                       div_valid_i,
  output logic                       div_ready_o,
  output logic                       clk_o,
  output logic [DIV_VALUE_WIDTH-1:0] cycl_count_o
);

  if ($clog2(DEFAULT_DIV_VALUE+1) > DIV_VALUE_WIDTH) begin : gen_elab_error
    initial $error("DEFAULT_DIV_VALUE (%0d) does not fit into DIV_VALUE_WIDTH (%0d)",
                   DEFAULT_DIV_VALUE, DIV_VALUE_WIDTH);
  end

  typedef enum logic [1:0] {IDLE, WAIT_LOW, LOAD_NEW} state_e;

  state_e                        state_q, state_d;
  logic [DIV_VALUE_WIDTH-1:0]    div_q, div_d;
  logic [DIV_VALUE_WIDTH-1:0]    cnt_q, cnt_d;
  logic                          clk_q, clk_d;
  logic                          bypass_q, bypass_d;
  logic                          en_q, en_d;

  function automatic logic [DIV_VALUE_WIDTH-1:0] half_period(input logic [DIV_VALUE_WIDTH-1:0] div);
    if (div <= 1) half_period = '0;
    else          half_period = div >> 1;
  endfunction

  function automatic logic [DIV_VALUE_WIDTH-1:0] low_period(input logic [DIV_VALUE_WIDTH-1:0] div);
    if (div <= 1) low_period = '0;
    else          low_period = div - half_period(div);
  endfunction

  always_comb begin
    state_d       = state_q;
    div_d         = div_q;
    cnt_d         = cnt_q;

          end else if (div_valid_i) begin

            end else begin
              state_d = WAIT_LOW;
            end
          end

              clk_d = clk_i;

              end else begin
                cnt_d = cnt_q + 1'b1;
              end
            end
          end
        end

          end else begin

      endcase
